cd <repo-folder>

git checkout -b my-feature

!stash
# Step 1: Clone the repository (skip if already cloned)
git clone <repository-url>
cd <repository-name>

# Step 2: Create or switch to a branch (e.g., main)
git checkout main

# Step 3: Make uncommitted changes to a file
echo "Temporary change" >> README.md

# Step 4: Stash the changes
git stash

# Step 5: Create and switch to another branch
git checkout -b temp-branch

# Step 6: Return to the original branch
git checkout main

# Step 7: Apply the stashed changes (but keep the stash)
git stash apply

# OR if you want to apply and remove the stash from stash list
# git stash pop

# Step 8: Stage and commit the changes
git add README.md
git commit -m "Applied stashed changes to README.md"

# Optional: Verify stash list (if applied instead of pop)
git stash list

# Step 9: Push changes if needed
git push origin main

!resolevmerge
# 1. Clone the repository
git clone https://github.com/archis04/check.git
cd check

# 2. Create and switch to branch-1
git checkout -b branch-1
echo "Line from branch-1" > conflict.txt
git add conflict.txt
git commit -m "Add line in branch-1"

# 3. Create and switch to branch-2
git checkout main
git checkout -b branch-2
echo "Line from branch-2" > conflict.txt
git add conflict.txt
git commit -m "Add line in branch-2"

# 4. Merge branch-2 into branch-1 to create a conflict
git checkout branch-1
git merge branch-2  # Will cause merge conflict

# 5. Open conflict in VS Code and resolve manually
# (Choose "Incoming + Current" to keep both lines)
# Then save the file

# 6. Add resolved file and continue
git add conflict.txt
git commit -m "Merge resolved manually"

# 7. Merge branch-1 into main
git checkout main
git merge branch-1

# 8. Push to remote (if fails, rebase first)
git push origin main  # Will likely fail due to remote changes

# 9. Rebase with remote and resolve conflicts again (if prompted)
git pull origin main --rebase

# 10. Resolve any conflict again → `git add conflict.txt`
git add conflict.txt
git rebase --continue

# 11. Finally, force push
git push origin main --force

!submodule
# Create main repository
mkdir main-repository
cd main-repository
git init

# Create submodule repository
mkdir submodule-repository
cd submodule-repository
git init
echo "This is the submodule repository" > file.txt
git add file.txt
git commit -m "Initial commit in submodule"
git remote add origin <submodule-repository-url>
git push -u origin master

# Add submodule to main repo
cd ../main-repository
git submodule add <submodule-repository-url> submodule
git commit -m "Added submodule"
git push origin master

# Initialize and update submodule
git submodule init
git submodule update

# Modify and commit changes in submodule
cd submodule
echo "Updated content in submodule" >> file.txt
git add file.txt
git commit -m "Updated content in submodule"
git push origin master

# Commit submodule changes in main repository
cd ..
git add submodule
git commit -m "Updated submodule reference"
git push origin master

!cherry-pick
# Clone the repository and create feature-branch
git clone https://github.com/archis04/main-repository.git
cd main-repository
git checkout -b feature-branch

# Make several commits in the feature-branch
echo "Feature 1" > feature.txt
git add feature.txt
git commit -m "First feature commit"
echo "Feature 2" >> feature.txt
git add feature.txt
git commit -m "Second feature commit"
echo "Feature 3" >> feature.txt
git add feature.txt
git commit -m "Third feature commit"

# Identify the commit hash (e.g., 4e5f6g for "Second feature commit")

# Switch to the main branch
git checkout main

# Cherry-pick the commit from feature-branch
git cherry-pick 4e5f6g  # Replace with actual commit hash

# If there are conflicts, resolve them, then commit
git add <conflicted_files>
git commit -m "Cherry-picked commit from feature-branch"

# Push changes to main branch
git push origin main
